function sharePost(index) { let posts = JSON.parse(localStorage.getItem("posts")) || []; let post = posts[index]; if (!post) { alert("Post not found"); return; } let message = `🔥 Check this out on Mzansi Market Place 🇿🇦 📌 ${post.title} 📖 ${post.description} 📞 Contact: ${post.phone} Visit here 👇 https://brachock.github.io/Mzansi-Market-place`; if (navigator.share) { navigator.share({ title: post.title, text: message, url: window.location.href }).catch(err => console.log("Share cancelled:", err)); } else if (/Android|iPhone|iPad|iPod/i.test(navigator.userAgent)) { let url = "https://wa.me/?text=" + encodeURIComponent(message); window.open(url, "_blank"); } else { navigator.clipboard.writeText(message).then(() => { alert("Post copied to clipboard!"); }).catch(() => { prompt("Copy this text to share:", message); }); } }
